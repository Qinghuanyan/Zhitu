(function () {
  const openBtn = document.getElementById("user-avatar-btn");
  if (!openBtn) return "missing-user-avatar-btn";
  openBtn.click();
  const resumeBtn = document.querySelector('#profile-tab-history [data-zhitu-action="resume_history_session"]');
  if (!resumeBtn) {
    return JSON.stringify({
      error: "missing-history-resume",
      overlayText: (document.getElementById("profile-overlay") && document.getElementById("profile-overlay").innerText || "").slice(0, 2000),
    });
  }
  resumeBtn.click();
  return JSON.stringify({
    currentTab: window.__ZHITU_STATE__ && window.__ZHITU_STATE__.currentTab,
    profileHidden: document.getElementById("profile-overlay") ? document.getElementById("profile-overlay").classList.contains("hidden") : null,
  });
})()
