(function () {
  const openBtn = document.getElementById("user-avatar-btn");
  if (openBtn) {
    openBtn.click();
  }
  const items = Array.from(
    document.querySelectorAll('#profile-tab-history [data-zhitu-action="resume_history_session"]')
  ).map((el, i) => ({
    index: i,
    conversationId: el.dataset.conversationId || "",
    text: ((el.parentElement && el.parentElement.innerText) || "").slice(0, 160),
  }));
  return JSON.stringify(items);
})()
