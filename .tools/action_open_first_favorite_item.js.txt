(function () {
  const btn = document.querySelector('#profile-tab-favorites [data-zhitu-action="open_favorite_item"]');
  if (!btn) return "missing-favorite-open";
  btn.click();
  return JSON.stringify({
    currentTab: window.__ZHITU_STATE__ && window.__ZHITU_STATE__.currentTab,
    profileHidden: document.getElementById("profile-overlay") ? document.getElementById("profile-overlay").classList.contains("hidden") : null,
  });
})()
