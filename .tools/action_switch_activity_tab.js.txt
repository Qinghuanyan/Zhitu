(function () {
  const btn = document.querySelector('.quick-tab[data-tab="activity"]');
  if (!btn) return "missing-activity-tab";
  btn.click();
  const chat = document.getElementById("chat-container");
  return JSON.stringify({
    currentTab: window.__ZHITU_STATE__ && window.__ZHITU_STATE__.currentTab,
    chatText: (chat && chat.innerText || "").slice(0, 1000),
  });
})()
