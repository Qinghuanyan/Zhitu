(function () {
  const btn = document.querySelector('.profile-tab[data-profile-tab="favorites"]');
  if (!btn) return "missing-profile-favorites-tab";
  btn.click();
  const overlay = document.getElementById("profile-overlay");
  return JSON.stringify({
    overlayHidden: overlay ? overlay.classList.contains("hidden") : null,
    text: (overlay && overlay.innerText || "").slice(0, 1500),
  });
})()
